# 1. Input mentah dari form                          (Laporan §5.2)

# 2. Cleaning milage                                 (Laporan §3.2)
milage = float(raw.replace(" mi.", "").replace(",", ""))   # 51000.0

# 3. Validasi — filter outlier                       (Laporan §3.2)
if not (0 <= milage <= 300_000):  ->  ditolak dengan pesan error

# 4. Prediksi dengan model .pkl                      (Laporan §5.2 / §3.7)
prediction = model.predict([[milage]])[0]             # 41229.32

# 5. Format output USD                               (Laporan §5.3)
return f"${prediction:,.2f}"                          # $41,229.32
